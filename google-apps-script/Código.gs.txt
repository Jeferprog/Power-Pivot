/**
 * Força a releitura do arquivo CSV no Google Drive (executa atualizarBase) e
 * devolve a base recém-gerada. É o que o botão "Sincronizar Agora" chama para
 * refletir IMEDIATAMENTE qualquer alteração no basecontascorrente.csv — ao
 * contrário de getBaseAgencias(), que só lê a planilha base já existente.
 *
 * Depende de atualizarBase(), obterPlanilhaBase() e CONFIG, definidos em
 * Atualizar_base.gs (mesmo projeto Apps Script).
 */
function forcarAtualizacaoBase() {
  atualizarBase();
  var aba = obterPlanilhaBase().getSheetByName(CONFIG.NOME_ABA);
  return aba.getDataRange().getDisplayValues();
}
